import sys
input = sys.stdin.readline

arr = []
for _ in range(9):
    arr.append(list(map(int, input().split())))

max = 0
for i in range(9):
    for j in range(9):
        if max < arr[i][j]:
            max = arr[i][j]
print(max)

for i in range(9):
    for j in range(9):
        if max == arr[i][j]:
            print(i+1, j+1)
            break


max 에 0 넣어놓고 그거보다 크면 값 넣고 
max값 찾은걸로 다시 포문 돌면서 찾으면 그 위치값 i, j 출력
